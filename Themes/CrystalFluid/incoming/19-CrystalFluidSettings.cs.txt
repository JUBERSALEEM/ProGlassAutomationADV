using System;
using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace CrystalFluid.Services
{
    /// <summary>
    /// Centralized Crystal Fluid design tokens accessible at runtime.
    /// Supports Theme Studio binding and reduced-motion preference.
    /// </summary>
    public sealed class CrystalFluidSettings : INotifyPropertyChanged
    {
        #region ── Singleton ──

        private static readonly Lazy<CrystalFluidSettings> _instance =
            new(() => new CrystalFluidSettings());

        public static CrystalFluidSettings Current => _instance.Value;

        private CrystalFluidSettings() { }

        #endregion

        #region ── Glass ──

        private double _glassOpacity = 1.0;
        public double GlassOpacity
        {
            get => _glassOpacity;
            set => Set(ref _glassOpacity, Clamp(value, 0, 1));
        }

        private double _glassBorderOpacity = 0.72;
        public double GlassBorderOpacity
        {
            get => _glassBorderOpacity;
            set => Set(ref _glassBorderOpacity, Clamp(value, 0, 1));
        }

        private double _reflectionOpacity = 0.80;
        public double ReflectionOpacity
        {
            get => _reflectionOpacity;
            set => Set(ref _reflectionOpacity, Clamp(value, 0, 1));
        }

        #endregion

        #region ── Water ──

        private double _waterOpacity = 1.0;
        public double WaterOpacity
        {
            get => _waterOpacity;
            set => Set(ref _waterOpacity, Clamp(value, 0, 1));
        }

        private double _waterSpeed = 1.0;
        public double WaterSpeed
        {
            get => _waterSpeed;
            set => Set(ref _waterSpeed, Clamp(value, 0.1, 3.0));
        }

        private double _waveAmplitude = 3.0;
        public double WaveAmplitude
        {
            get => _waveAmplitude;
            set => Set(ref _waveAmplitude, Clamp(value, 0, 10));
        }

        private double _causticIntensity = 0.70;
        public double CausticIntensity
        {
            get => _causticIntensity;
            set => Set(ref _causticIntensity, Clamp(value, 0, 1));
        }

        private double _bubbleIntensity = 0.70;
        public double BubbleIntensity
        {
            get => _bubbleIntensity;
            set => Set(ref _bubbleIntensity, Clamp(value, 0, 1));
        }

        #endregion

        #region ── Shadow ──

        private double _shadowOpacity = 0.35;
        public double ShadowOpacity
        {
            get => _shadowOpacity;
            set => Set(ref _shadowOpacity, Clamp(value, 0, 1));
        }

        #endregion

        #region ── Corner Radius ──

        private double _cornerRadiusLarge = 30;
        public double CornerRadiusLarge
        {
            get => _cornerRadiusLarge;
            set => Set(ref _cornerRadiusLarge, Clamp(value, 0, 50));
        }

        #endregion

        #region ── Motion ──

        private bool _reducedMotion = false;
        public bool ReducedMotion
        {
            get => _reducedMotion;
            set => Set(ref _reducedMotion, value);
        }

        private double _animationIntensity = 1.0;
        public double AnimationIntensity
        {
            get => _animationIntensity;
            set => Set(ref _animationIntensity, Clamp(value, 0, 2));
        }

        #endregion

        #region ── Helpers ──

        private static double Clamp(double value, double min, double max)
            => Math.Max(min, Math.Min(max, value));

        #endregion

        #region ── INotifyPropertyChanged ──

        public event PropertyChangedEventHandler PropertyChanged;

        private void OnPropertyChanged([CallerMemberName] string name = null)
            => PropertyChanged?.Invoke(this,
                new PropertyChangedEventArgs(name));

        private void Set<T>(ref T field, T value,
            [CallerMemberName] string name = null)
        {
            if (Equals(field, value)) return;
            field = value;
            OnPropertyChanged(name);
        }

        #endregion
    }
}
